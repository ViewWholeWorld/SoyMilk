```js
import { spawn } from 'node:child_process';
import { mkdir, writeFile, rm } from 'node:fs/promises';
import { request, createServer } from 'node:http';
import assert from 'node:assert/strict';
import { once } from 'node:events';
const fixture = 'apps/web/build/client-review-fixture';
await mkdir(fixture, { recursive: true });
await writeFile(`${fixture}/probe.txt`, 'review-only-sibling-sentinel');
const sockets = new Set();
let upstreamSeen, upstreamClosed = false;
const seen = new Promise(resolve => { upstreamSeen = resolve; });
const api = createServer((req, res) => {
  res.on('close', () => { upstreamClosed = true; });
  upstreamSeen();
});
api.on('connection', s => { sockets.add(s); s.on('close', () => sockets.delete(s)); });
api.listen(0, '127.0.0.1'); await once(api, 'listening');
const child = spawn(process.execPath, ['apps/web/server.ts'], { env: { ...process.env, WEB_PORT: '34193', WEB_HOST: '127.0.0.1', API_BASE_URL: `http://127.0.0.1:${api.address().port}` }, stdio: ['ignore', 'pipe', 'pipe'] });
try {
  await new Promise((resolve, reject) => {
    const timer = setTimeout(() => reject(new Error('web start timeout')), 15000);
    child.stdout.on('data', data => { if (data.toString().includes('web started')) { clearTimeout(timer); resolve(); } });
    child.on('exit', code => { clearTimeout(timer); reject(new Error(`web exited ${code}`)); });
  });
  const result = await new Promise((resolve, reject) => {
    request({ host: '127.0.0.1', port: 34193, path: '/%2e%2e/client-review-fixture/probe.txt' }, res => {
      let text = ''; res.on('data', chunk => { text += chunk; }); res.on('end', () => resolve({ status: res.statusCode, text }));
    }).on('error', reject).end();
  });
  assert.deepEqual(result, { status: 200, text: 'review-only-sibling-sentinel' });
  console.log('R09 confirmed: encoded parent path returns same-prefix sibling sentinel, HTTP 200');
  const req = request({ host: '127.0.0.1', port: 34193, path: '/api/v1/items' });
  req.on('error', () => {}); req.end();
  await seen; req.destroy();
  await new Promise(resolve => setTimeout(resolve, 1500));
  assert.equal(upstreamClosed, false);
  console.log('R10 confirmed: client disconnected but upstream response still open after 1500ms');
} finally {
  child.kill('SIGKILL');
  for (const s of sockets) s.destroy();
  api.close();
  await rm(fixture, { recursive: true, force: true });
}
```
